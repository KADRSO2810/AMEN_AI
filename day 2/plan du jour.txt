convertitr le tableau d'une pension livrée en dataframe et l'enregistrer dans un fichier .csv






import os
import pandas as pd
import fitz  # PyMuPDF
from datetime import datetime
import re

def find_pension_page(doc):
    """Trouve la page contenant la table de pension livrée"""
    for page_num, page in enumerate(doc):
        text = page.get_text()
        if "Pension Livrée" in text and "Les opérations du jour" in text:
            return page_num
    return None

def extract_pension_table(page):
    """Extrait la table de pension de la page spécifique"""
    tabs = page.find_tables()
    if not tabs:
        return None
    
    # Essayer toutes les tables de la page jusqu'à trouver la bonne
    for table in tabs:
        data = table.extract()
        
        # Vérifier si c'est bien notre table
        if len(data) > 1 and len(data[0]) >= 4:  # Au moins 4 colonnes
            # Vérifier la présence de colonnes attendues
            headers = [str(h).strip().lower() for h in data[0]]
            expected = ['isin', 'libellé', 'montant', 'taux']
            if sum(1 for e in expected if any(e in h for h in headers)) >= 3:
                return data
    
    return None

def clean_data(table_data):
    """Nettoie et structure les données extraites"""
    # Récupérer les en-têtes et données
    headers = [str(h).strip() for h in table_data[0]]
    rows = table_data[1:]
    
    # Créer le DataFrame
    df = pd.DataFrame(rows, columns=headers)
    
    # Standardiser les noms de colonnes
    col_mapping = {
        'isin': 'ISIN',
        'libellé': 'Libellé',
        'nombre de titres': 'Nombre de Titres',
        'montant': 'Montant',
        'échéance': 'Échéance',
        'taux': 'Taux'
    }
    
    # Renommer les colonnes
    for col in df.columns:
        col_lower = col.lower()
        for target, replacement in col_mapping.items():
            if target in col_lower:
                df = df.rename(columns={col: replacement})
                break
    
    # Supprimer les lignes vides ou non pertinentes
    df = df.dropna(how='all')
    df = df[~df.iloc[:, 0].str.contains('Les opérations|Montant Global', na=False)]
    
    return df

def extract_pension_livree(pdf_path):
    """Fonction principale d'extraction"""
    try:
        doc = fitz.open(pdf_path)
        
        # Trouver la bonne page
        page_num = find_pension_page(doc)
        if page_num is None:
            print("Page contenant la table de pension non trouvée")
            return None
        
        # Extraire la table
        table_data = extract_pension_table(doc[page_num])
        if table_data is None:
            print("Table de pension non trouvée sur la page")
            return None
        
        # Nettoyer les données
        return clean_data(table_data)
    
    except Exception as e:
        print(f"Erreur lors de l'extraction: {str(e)[:200]}")
        return None

def main():
    # Configuration
    input_pdf = "C:/Users/zizou/OneDrive/Desktop/stage 3ème/day 2/pdffiles/Bulletin quotidien des activitées.pdf"
    output_folder = "C:/Users/zizou/OneDrive/Desktop/stage 3ème/day 2/csvfiles/"
    
    # Extraction
    print(f"\nTraitement du fichier: {os.path.basename(input_pdf)}")
    df = extract_pension_livree(input_pdf)
    
    if df is not None and not df.empty:
        # Sauvegarde
        os.makedirs(output_folder, exist_ok=True)
        timestamp = datetime.now().strftime('%Y%m%d_%H%M%S')
        output_path = os.path.join(output_folder, f"pension_livree_{timestamp}.csv")
        df.to_csv(output_path, index=False, encoding='utf-8-sig')
        
        print("Extraction réussie!")
        print(f"Fichier sauvegardé: {output_path}")
        print("\nAperçu des données:")
        print(df.head())
    else:
        print("Échec de l'extraction - aucune donnée valide trouvée")

if __name__ == "__main__":
    main()
